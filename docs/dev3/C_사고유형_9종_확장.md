## 4. 가중치 자동 갱신 스크립트 가이드

`backend/scripts/expand_weights_to_9.py` 권장 구조:

```python
"""weights.json 의 모든 점수 dict 에 신규 4 사고유형 키를 0 으로 채움.
   학술 근거가 명확한 항목은 별도 dict 로 실제 가중치 부여."""
import json

NEW_TYPES = ["RANSOMWARE", "MISCONFIG_EXPOSURE", "PHISH_INITIAL_ACCESS", "INSIDER_THREAT"]

# 실제 가중치 (학술 근거 있는 항목만 수기 정의)
ACTUAL_BOOSTS = {
    "Q1_inception:service_anomaly": {"RANSOMWARE": 25},  # T1485/T1486 Impact 단계
    "Q2D:data_destruction": {"RANSOMWARE": 30},
    "Q2B:data_exposed": {"MISCONFIG_EXPOSURE": 25},
    "Q3:storage_policy_change": {"MISCONFIG_EXPOSURE": 25},
    "Q1_inception:external_report": {"PHISH_INITIAL_ACCESS": 10},
    "Q2B:identity_theft": {"PHISH_INITIAL_ACCESS": 25},
    "Q3:fail_then_success": {"PHISH_INITIAL_ACCESS": 15},
    "Q1_inception:audit": {"INSIDER_THREAT": 10},
    "Q2C:permission_changes": {"INSIDER_THREAT": 20},
    "ACCOUNT:newly_created": {"INSIDER_THREAT": 15},
    "T1486": {"RANSOMWARE": 35},
    "T1490": {"RANSOMWARE": 25},
    "T1190": {"MISCONFIG_EXPOSURE": 30},
    "T1566": {"PHISH_INITIAL_ACCESS": 35},
    "T1078": {"INSIDER_THREAT": 20},
    # ... (학술 근거 부착 필수)
}

# 신규 ATT&CK 기법 추가 (weights.json partial_analysis_techniques 에 신규 키)
NEW_TECHNIQUES = {
    "T1486": {"tactic": "Impact", "source": "MITRE Data Encrypted for Impact"},
    "T1490": {"tactic": "Impact", "source": "MITRE Inhibit System Recovery"},
    "T1190": {"tactic": "Initial Access", "source": "MITRE Exploit Public-Facing Application"},
    "T1566": {"tactic": "Initial Access", "source": "MITRE Phishing"},
}

def expand():
    with open("backend/data/triage/weights.json", encoding="utf-8") as f:
        w = json.load(f)
    # 각 매트릭스에 신규 키 0 채움
    for section in ["decision_tree_answers", "clue_types", "partial_analysis_techniques", "environment_modifiers_to_hypothesis"]:
        for k, v in w[section].items():
            for nt in NEW_TYPES:
                v.setdefault(nt, 0)
    # 실제 가중치 부여
    for key, boosts in ACTUAL_BOOSTS.items():
        for section in ["decision_tree_answers", "clue_types", "partial_analysis_techniques", "environment_modifiers_to_hypothesis"]:
            if key in w[section]:
                for nt, score in boosts.items():
                    w[section][key][nt] = score
                break
    # 신규 ATT&CK 기법 추가
    for tid, meta in NEW_TECHNIQUES.items():
        if tid not in w["partial_analysis_techniques"]:
            w["partial_analysis_techniques"][tid] = {
                **{nt: 0 for nt in NEW_TYPES + ["CREDENTIAL_THEFT", "PRIVILEGE_ESCALATION", "DATA_EXFILTRATION", "AUDIT_LOG_TAMPERING", "MULTI_CSP_LATERAL"]},
                **meta,
            }
            # 신규 기법별 실제 가중치는 ACTUAL_BOOSTS 에서 처리
            for nt, score in ACTUAL_BOOSTS.get(tid, {}).items():
                w["partial_analysis_techniques"][tid][nt] = score
    with open("backend/data/triage/weights.json", "w", encoding="utf-8") as f:
        json.dump(w, f, ensure_ascii=False, indent=2)
    print("OK — weights.json expanded to 9 incident types")

if __name__ == "__main__":
    expand()
```

→ 가중치 부여는 **수기 검토 필수** (단위 5 정책 + 학술 근거 부착).
